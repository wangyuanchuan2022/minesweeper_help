def make_solver(w=5, h=5, a=3, is_play=False):
    """构造一个「轻量」Solver 实例，用于不发射信号的纯逻辑方法。

    使用 ``Solver.__new__`` 跳过 ``__init__``（避免读取 cfg.json 与加载模板图片），
    并手动初始化被测方法所需的属性。
    """
    solver = Solver.__new__(Solver)
    solver.w = w
    solver.h = h
    solver.a = a
    solver.is_play = is_play
    solver.cell_width = 24
    solver.screenshot_w = int(solver.cell_width * 5 / 9)
    solver.screenshot_h = int(solver.cell_width * 7 / 9)
    solver.pos_dict_list = []
    solver.appended_pos = set()
    solver.num = 0
    solver.checked = {}
    solver.memory = {}
    solver.cell_value = None
    solver.img = None
    solver.bx = 0
    solver.by = 0
    solver._bx = 0
    solver._by = 0
    solver.count = 0
    solver.p = (a / (w * h)) if (w * h) else 0.0
    solver.images = None
    solver._images_cell_width = None
    solver._images_mtime = None
    solver._locate_templates = {}
    solver.stats_data = None
    solver._stats_updates = 0
    return solver
侧门钥匙：Solver.__new__(Solver) 造出对象但不跑 __init__——绕过大门：不读 cfg.json、不加载 25 张模板图片、不初始化 Qt 线程信号。为什么必须绕：__init__ 的每一步都依赖外部世界（文件存在、图片格式正确、Qt 环境），任何一步坏掉，71 个测试会成片崩掉；单元测试要的是「只测被测方法」。
摆实验台：手动把被测方法用到的属性一个个摆好——几何（w/h/cell_width/切片尺寸，与真实默认值一致）、推理状态（checked/memory/pos_dict_list）、统计缓存占位。对照 utils/solver.py L53-L93 的 __init__ 正是这份清单的「正门版」；连 p 都按公式手工算（L39，m11 字段表的派生项）。测试与生产走同一套属性约定，只是来源不同。

这把侧门钥匙换来的东西看得很清楚：71 个测试全部只依赖内存里摆好的属性，不碰磁盘、不碰屏幕。它们都测什么？

分布本身就是架构的体检报告：越纯的层（组合数学）测试越多、越易测；越集成的层（概率引擎要真实棋盘数据）测试越少但越重。HANDOFF 文档把验收线写死了：§3 的验证结果是 71 个测试全部通过（exit 0），§5 关键约束第 3 条要求每个任务完成后测试套件必须全绿。
